"""Offline audit probes: execute bounded workflow fragments with fake registries.

No credentials, network, publication, or Git mutations. Run from the repo root.
These assertions preserve the audited failures, not desired production behavior.
"""
import hashlib
import importlib.util
import json
from pathlib import Path
import subprocess
import tempfile
import textwrap

root = Path.cwd()
workflow = root / '.github/workflows/reconcile-release.yml'
assert hashlib.sha256(workflow.read_bytes()).hexdigest() == '9b705b5ba4c203d68c7378828bfca195ad0328ea2db41010a71c69e5dd9fc955', 'Replay only the audited workflow snapshot'
lines = workflow.read_text().splitlines()


def fragment(first, last):
    return textwrap.dedent('\n'.join(lines[first - 1:last]))


def shell(body):
    result = subprocess.run(['bash', '-c', body], capture_output=True, text=True)
    return {'exit_code': result.returncode, 'stdout': result.stdout.strip()}


records = []
missing = shell('npm() { return 1; }; package_name=synthetic-package; '
                'package_version=synthetic-version; SOURCE_REVISION=synthetic-source;\n'
                + fragment(178, 183) + '\nprintf "recovery-reached"\n')
assert missing['exit_code'] == 1
records.append({'id': 'R1', 'scenario': 'wrapper absent; another registry succeeded',
                'workflow_lines': [178, 183], 'observed': missing})

existing = shell('npm() { return 0; }; name=synthetic-package; version=synthetic-version;\n'
                 'probe() {\n' + fragment(243, 246)
                 + '\nprintf "content-verification-reached";\n}; probe\n')
assert existing['exit_code'] == 0 and 'content-verification-reached' not in existing['stdout']
records.append({'id': 'R2', 'scenario': 'dependency version exists with conflicting bytes',
                'workflow_lines': [243, 246], 'observed': existing})

with tempfile.TemporaryDirectory(prefix='redact-audit-probe-') as directory:
    output = Path(directory) / 'outputs'
    partial = shell('curl() { printf "200"; }; name=synthetic-project; '
                    'version=synthetic-version; GITHUB_OUTPUT=' + str(output) + ';\n'
                    + fragment(386, 391) + '\nprintf "file-inventory-comparison-reached"\n')
    assert partial['exit_code'] == 0 and 'needs_publish=false' in output.read_text()
    records.append({'id': 'R3', 'scenario': 'PyPI version exists with only one required file uploaded',
                    'workflow_lines': [386, 391], 'observed': partial,
                    'github_output': output.read_text().strip()})

spec = importlib.util.spec_from_file_location('audit_manifest', root / 'scripts/release-manifest.py')
module = importlib.util.module_from_spec(spec)
spec.loader.exec_module(module)
try:
    module.build_manifest(source_revision='synthetic-source', conformance_identity='synthetic-corpus',
                          version='synthetic-version', artifact_set=[], registry_state={})
except ValueError as error:
    records.append({'id': 'R4', 'scenario': 'qualification fails; all publish jobs skipped',
                    'workflow_lines': [692, 727], 'workflow': '.github/workflows/release.yml',
                    'observed': {'manifest_created': False, 'error': str(error)}})
else:
    raise AssertionError('Expected empty-state manifest failure at audited revision')

partial_manifest = module.build_manifest(
    source_revision='synthetic-source', conformance_identity='synthetic-corpus',
    version='synthetic-version', artifact_set=['npm:synthetic-package'],
    registry_state={'npm:synthetic-package': 'published'})
assert len(partial_manifest['artifact_set']) == 1
records.append({'id': 'R5', 'scenario': 'only one publish job emits registry state',
                'workflow': '.github/workflows/release.yml', 'workflow_lines': [692, 727],
                'observed': {'manifest_artifact_count': len(partial_manifest['artifact_set']),
                             'missing_artifacts_recorded_as_unknown': False}})
print(json.dumps({'source_revision': subprocess.check_output(['git', 'rev-parse', 'HEAD'], text=True).strip(),
                  'probes': records}, indent=2))
